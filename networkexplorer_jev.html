library(glue)
source("simulation_jev/simulation_jev_functions.r")

# Load and prepare data

statements_text <- read_parquet("parquet/statements_meta.parquet") |>
  select(id = S_id, text = Worldview_Statement)
dec <- read_parquet("MyFirstJevSim_backup.parquet")
mem <- decisions_to_memory(dec)

# Get list of all time steps to process
time_steps <- c(5, 10, 20, 50, 100)

# Create JSONs for each time step
manifest_data <- list()

for (t in time_steps) {
  mem_t <- mem |> filter(time_step == t)

  if (nrow(mem_t) == 0) {
    message("Skipping t=", t, " (no data)")
    next
  }

  # Get statements present at this time step
  statements_t <- mem_t |>
    select(memory) |>
    distinct() |>
    rename(id = memory) |>
    left_join(statements_text, by = "id") |>
    mutate(
      label = coalesce(text, id), # Use statement text as label; fall back to id
      type = "statement"
    ) |>
    select(id, label, type)

  # Get agents present at this time step
  agents_t <- mem_t |>
    select(agent_id) |>
    distinct() |>
    mutate(
      id = paste0("A", agent_id),
      label = id,
      type = "agent"
    ) |>
    select(id, label, type)

  # Prepare nodes
  nodes <- bind_rows(agents_t, statements_t) |>
    mutate(across(everything(), as.character)) |>
    mutate(across(everything(), \(x) replace_na(x, "")))

  # Prepare edges
  edges <- mem_t |>
    mutate(
      source = paste0("A", agent_id),
      target = memory,
      type = "AS",
      weight = 1,
      .keep = "none"
    ) |>
    distinct()

  message("t=", t, " | Nodes: ", nrow(nodes), " | Edges: ", nrow(edges))

  # Write JSON
  graph_json <- toJSON(
    list(nodes = nodes, links = edges),
    auto_unbox = TRUE,
    pretty = FALSE
  )

  json_file <- glue("networkjson/jev_t{t}.json")
  dir.create("networkjson", showWarnings = FALSE)
  write_file(graph_json, json_file)

  # Add to manifest
  manifest_data[[length(manifest_data) + 1]] <- list(
    file = json_file,
    t = t,
    N = n_distinct(mem_t$agent_id),
    M = 10,
    num_statements = n_distinct(mem_t$memory)
  )
}
